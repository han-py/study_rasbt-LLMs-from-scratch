## 📦 第一步：准备“岗前培训”的教材 (代码清单 7-1)

模型要学规矩，我们得给它成千上万份标准的“提问+答案”示范。这段代码的作用就是一个**跑腿员**，从网上把带有五万道题的 `instruction-data.json` 题库下载到电脑里。

```python
import json
import os
import urllib

def download_and_load_file(file_path, url):
    # 检查脑电里是不是已经有这个文件了
    if not os.path.exists(file_path):
        # 没有的话，顺着网线请求下载
        with urllib.request.urlopen(url) as response:
            test_data = response.read().decode("utf-8")
        # 把下载回来的数据写进本地文件
        with open(file_path, "w", encoding="utf-8") as file:
            file.write(test_data)
    
    # 把存在硬盘里的文本文件读取出来，变成 Python 能直接操作的“字典（列表）”格式
    with open (file_path, "r") as file:
        data = json.load(file)
    return data

file_path = "instruction-data.json"
url = "http://raw.githubusercontent.com/rasbt/LLMs-from-scratch/main/ch07/01_main-chapter-code/instruction-data.json"

data = download_and_load_file(file_path, url)
```

## 🏷️ 第二步：将散装数据拼装成“模板剧本” (代码清单 7-2 & 7-3)

AI 缺乏人类看懂隐藏语境的能力。如果直接给它题目，它会一头雾水。所以我们需要一个 `format_input` 函数，强制把下载来的散装数据套进刻板但清晰的“角色扮演格式”里。然后把整个大题库按比例切分。

```python
def format_input(entry):
    # 【给AI强行加戏】给它立一个人设，告诉它当前在干嘛
    instruction_text = (
        f"Below is an instruction that describes a task. "
        f"Write a response that appropriately completes the request."
        f"\n\n### Instruction:\n{entry['instruction']}"
    )

    # 【补充输入材料】比如翻译任务，这里就是等被翻译的原文。如果没有就留空。
    input_text = (
        f"\n\n### Input:\n{entry['input']}" if entry["input"] else  ""
    )
    # 把立的人设和题目像积木一样拼接返回
    return instruction_text + input_text

# ============ 划分数据集 ============
train_portion = int(len(data) * 0.85) # 85% 题量：平时练习卷（模型看着答案学）
test_portion = int(len(data) * 0.1)   # 10% 题量：期末摸底卷（绝对保密，用来考它）
val_portion = len(data) - train_portion - test_portion # 5% 题量：阶段测验卷

# 利用切片把数据集一分为三
train_data = data[:train_portion]
test_data = data[train_portion:train_portion + test_portion]
val_data = data[train_portion + test_portion:]
```

## 🔢 第三步：把文字变成巨型数学矩阵 (代码清单 7-4 & 7-5)

这是训练大模型最难的数据处理部分！GPU 只吃规规矩矩的数字组成的“长方形矩阵”。但人类的话有长有短。
* 通过 `InstructionDataset`，我们先把中英文翻译成数字编号 (Token IDs)。
* 通过 `custom_collate_fn` 这个“发卷打包员”，我们把所有短考卷强制补充空格 (`50256`)，使其变得跟这批考卷里最长的那份一样长！

```python
import torch
from torch.utils.data import Dataset
import tiktoken

tokenizer = tiktoken.get_encoding("gpt2") # 召唤 GPT-2 的官方数字密码本

class InstructionDataset(Dataset):
    def __init__(self, data, tokenizer):
        self.data = data
        self.encoded_texts = []
        for entry in data:
            # 把题面和正确答案用 "### Response:" 隔开，拼成超级长的一段话
            instruction_plus_input = format_input(entry)
            response_text = f"\n\n### Response:\n{entry['output']}"
            full_text = instruction_plus_input + response_text
            
            # encode核心操作：把这段话里的字母全部替换成 [115, 654, 888] 这样的数字
            self.encoded_texts.append(tokenizer.encode(full_text))

    def __getitem__(self, index):
        return self.encoded_texts[index]
    def __len__(self):
        return len(self.data)

# ======= 究极“截长补短”打包函数 =======
def custom_collate_fn(batch, pad_token_id=50256, ignore_index=-100, allowed_max_length=1024, device="cpu"):
    # 找到这批卷子里最长的那份 (打个比方，最长是50个字)
    batch_max_length = max(len(item)+1 for item in batch)
    inputs_lst, targets_lst = [], []

    for item in batch:
        new_item = item.copy()
        new_item += [pad_token_id] # 结尾补上终止符

        # 【补白操作】：只要卷子不到50个字，就疯狂往屁股后面垫 50256
        padded = (new_item + [pad_token_id] * (batch_max_length - len(new_item)))
        
        # 【错位游戏】：输入切掉尾巴，目标切掉开头，刚好形成错位1格去“预测下一个字”
        inputs = torch.tensor(padded[:-1]) 
        targets = torch.tensor(padded[1:]) 

        # 🌟【重磅！免死金牌机制】：
        # 我们不能让 AI 浪费脑力去记忆怎么生成无聊的填充空白符(50256)
        # 所以算出哪里是 50256，把答案区的它们强行改成 -100。
        # PyTorch 的交叉熵计分器只要看到 -100，就不加分也不扣分！
        mask = targets == pad_token_id
        indices = torch.nonzero(mask).squeeze()
        if indices.numel() > 1:
            targets[indices[1:]] = ignore_index

        inputs_lst.append(inputs)
        targets_lst.append(targets)

    # 堆叠成长方形的数学矩阵，送入显卡计算
    return torch.stack(inputs_lst).to(device), torch.stack(targets_lst).to(device)
```

## 👨‍⚖️ 第六步：利用强AI去自动化审阅小AI (代码清单 7-10 & 7-11)

模型特训结业！它吭哧吭哧答完了几百道保密考题（测评分保存在一个文件里）。我们要利用另一个强力开源系统 `Ollama (Llama-3)` 充当阅卷老师打分。

```python
import urllib.request
import json

def query_model(prompt, model="llama3", url="http://localhost:11434/api/v1/chat"):
    """这是对讲机：专门发邮件给本机后台挂着的全能博士 Llama3，求它帮忙打分"""
    
    # 按照 Ollama 的规矩，把咱们的“打分请求”装箱
    data = {
        "model": model,
        "messages": [{"role": "user", "content": prompt}],
        "options": { "seed": 123, "temperature": 0 } # 温度0：代表必须极度铁面无私
    }

    # 把包裹打包装好并编码发送
    payload = json.dumps(data).encode("utf-8")
    request = urllib.request.Request(url, data=payload, method="POST")
    request.add_header("Content-Type", "application/json")

    # 苦苦等待 Llama3 审阅完成，并接收它吐出的文字
    response_data = ""
    with urllib.request.urlopen(request) as response:
        while True:
            line = response.readline().decode("utf-8")
            if not line: break
            response_json = json.loads(line)
            response_data += response_json["message"]["content"]

    return response_data

# === 流水线自动化大判卷 ===
def generate_model_scores(json_data, json_key, model="llama3"):
    scores = []
    # 从几百道测试题记录里轮流抽一道出来
    for entry in tqdm(json_data, desc="Scoring entries"):
        
        # 构建给裁判的提词：“下面左边是小弟的破答案，右边是教科书好答案，请你给出0~100的分数”
        prompt = (
            f"Given the input `{format_input(entry)}` "
            f"and correct output `{entry['output']}`, "
            f"score the model response `{entry[json_key]}`"
            f" on a scale from 0 to 100, where 100 is the best score. "
            f"Respond with the integer number only." 
        )
        
        # 呼叫上面的发邮件函数，拿到 Llama3 给出的整数分数
        score = query_model(prompt, model=model)
        try:
            scores.append(int(score))
        except ValueError:
            continue
    return scores

# 最终计算并汇报大模型的平均战斗力！

print(f"Average score: {sum(scores) / len(scores):.2f}\n")
```
